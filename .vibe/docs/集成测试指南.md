### 2.1 CommandBuilder 测试

测试文件: `src/ffmpeg/services/__tests__/commandBuilder.spec.ts`

#### 测试用例

```typescript
describe('CommandBuilder', () => {
  describe('build', () => {
    it('should generate basic command', async () => {
      const preset = createBasicPreset();
      const command = await commandBuilder.build(
        preset,
        'input.mp4',
        'output.mp4'
      );

      expect(command).toContain('ffmpeg -i input.mp4');
      expect(command).toContain('-c:v libx264');
      expect(command).toContain('-c:a aac');
      expect(command).toContain('output.mp4');
    });

    it('should handle H.265 encoder', async () => {
      const preset = createH265Preset();
      const command = await commandBuilder.build(preset, 'input.mp4', 'output.mp4');

      expect(command).toContain('libx265');
    });

    it('should add filters correctly', async () => {
      const preset = createPresetWithFilters();
      const command = await commandBuilder.build(preset, 'input.mp4', 'output.mp4');

      expect(command).toContain('-vf');
    });

    it('should handle trimming', async () => {
      const preset = createPresetWithTrimming();
      const command = await commandBuilder.build(preset, 'input.mp4', 'output.mp4');

      expect(command).toContain('-ss 00:00:10');
      expect(command).toContain('-to 00:00:20');
    });

    it('should handle two-pass encoding', async () => {
      const preset = createTwoPassPreset();
      const command = await commandBuilder.build(preset, 'input.mp4', 'output.mp4');

      expect(command).toContain('-pass 1');
      expect(command).toContain('-pass 2');
    });
  });

  describe('validateCommand', () => {
    it('should validate valid command', () => {
      const command = 'ffmpeg -i input.mp4 -c:v libx264 output.mp4';
      const result = commandBuilder.validateCommand(command);

      expect(result.valid).toBe(true);
    });

    it('should detect invalid command', () => {
      const command = 'ffmpeg -i input.mp4 -invalid-param output.mp4';
      const result = commandBuilder.validateCommand(command);

      expect(result.valid).toBe(false);
      expect(result.errors.length).toBeGreaterThan(0);
    });
  });
});
```

### 2.2 ProgressParser 测试

```typescript
describe('ProgressParser', () => {
  describe('parse', () => {
    it('should parse progress line', () => {
      const line = 'frame= 324 fps= 30 q=23.0 size= 250MB time=00:15:30.45 bitrate=5200kbits/s speed=1.8x';
      const progress = progressParser.parse(line);

      expect(progress).not.toBeNull();
      expect(progress?.frame).toBe(324);
      expect(progress?.fps).toBe(30);
      expect(progress?.q).toBe(23);
      expect(progress?.size).toBe(250 * 1024 * 1024);
      expect(progress?.bitrate).toBe(5200);
      expect(progress?.speed).toBe(1.8);
    });

    it('should calculate progress percentage', () => {
      const output = 'Duration: 00:30:00.00\nframe= 324 fps= 30 time=00:15:30.45';
      const progress = progressParser.parseFull(output);

      expect(progress.progress).toBeCloseTo(50, 0);
    });

    it('should estimate remaining time', () => {
      const progress = {
        timeSeconds: 900,
        durationSeconds: 1800,
        speed: 2.0
      };

      const remaining = progressParser.estimateRemainingTime(progress);

      expect(remaining).toBeCloseTo(450, 0);
    });
  });
});
```
