## 一、设计概述

### 1.1 设计目标

本文档详细描述 FFmpegFreeUI (3FUI) 融合功能的模块化设计，包括四个核心模块：

1. **FFmpeg 参数生成模块** (阶段 1)
2. **进度解析模块** (阶段 2)
3. **预设管理模块** (阶段 3)
4. **高级功能模块** (阶段 4)

### 1.2 设计原则

- **模块化**: 各模块职责清晰，接口明确
- **可扩展**: 易于添加新功能和编码器
- **可测试**: 单元测试友好，依赖注入
- **兼容性**: 与现有 Rebebuca 架构无缝集成

### 1.3 技术栈

| 层级 | 技术选型 | 说明 |
|------|---------|------|
| **类型定义** | TypeScript | 类型安全，减少运行时错误 |
| **状态管理** | Pinia (Composition API) | 响应式状态管理 |
| **UI 组件** | Vue 3 + Naive UI | 现代化组件库 |
| **命令执行** | Node.js + PTY | 跨平台终端模拟 |
| **数据存储** | Node.js Store | 本地持久化存储 |

### 3.4 核心类实现

#### 3.4.1 CommandBuilder 类

```typescript
// src/ffmpeg/services/commandBuilder.ts

import type { FFmpegPreset } from '../types/preset';
import { ValidationService } from './validationService';
import { escapeShellArg, normalizePath } from '../utils/path';
import { TimeUtils } from '../utils/time';

/**
 * FFmpeg 命令行生成器
 * 参考自 FFmpegFreeUI/编码任务/预设管理.vb
 */
export class CommandBuilder {
  private validationService: ValidationService;

  constructor(validationService?: ValidationService) {
    this.validationService = validationService || new ValidationService();
  }

  /**
   * 构建完整的 FFmpeg 命令行
   * @param preset FFmpeg 预设配置
   * @param inputFile 输入文件路径
   * @param outputFile 输出文件路径
   * @returns 完整的 FFmpeg 命令行字符串
   */
  async build(
    preset: FFmpegPreset,
    inputFile: string,
    outputFile: string
  ): Promise<string> {
    // 验证预设
    const validationResult = this.validationService.validatePreset(preset);
    if (!validationResult.valid) {
      throw new Error(`Invalid preset: ${validationResult.errors.join(', ')}`);
    }

    const args: string[] = [];

    // ==================== 全局参数 ====================
    args.push('-hide_banner', '-nostdin', '-y');

    // ==================== 解码参数 ====================
    this.addDecoderArgs(args, preset);

    // ==================== 剪辑区间 ====================
    this.addTrimmingArgs(args, preset);

    // ==================== 自定义前置参数 ====================
    if (preset.custom.preParams) {
      args.push(...this.parseCustomParams(preset.custom.preParams));
    }

    // ==================== 输入文件 ====================
    args.push('-i', escapeShellArg(normalizePath(inputFile)));

    // ==================== 视频编码参数 ====================
    if (preset.video.enabled) {
      const videoArgs = this.buildVideoArgs(preset);
      args.push(...videoArgs);
    }

    // ==================== 音频编码参数 ====================
    if (preset.audio.enabled) {
      const audioArgs = this.buildAudioArgs(preset);
      args.push(...audioArgs);
    }

    // ==================== 流映射 ====================
    const mapArgs = this.buildStreamMap(preset);
    args.push(...mapArgs);

    // ==================== 元数据 ====================
    if (preset.metadata) {
      this.addMetadataArgs(args, preset.metadata);
    }

    // ==================== 自定义后置参数 ====================
    if (preset.custom.postParams) {
      args.push(...this.parseCustomParams(preset.custom.postParams));
    }

    // ==================== 输出文件 ====================
    args.push(escapeShellArg(normalizePath(outputFile)));

    return args.join(' ');
  }

  /**
   * 构建视频编码参数
   */
  private buildVideoArgs(preset: FFmpegPreset): string[] {
    const args: string[] = [];

    // 编码器
    if (preset.video.encoder) {
      args.push('-c:v', preset.video.encoder);
    }

    // 预设
    if (preset.video.preset) {
      args.push('-preset', preset.video.preset);
    }

    // 配置文件
    if (preset.video.profile) {
      args.push('-profile:v', preset.video.profile);
    }

    // 级别
    if (preset.video.level) {
      args.push('-level', preset.video.level);
    }

    // 调整
    if (preset.video.tune) {
      args.push('-tune', preset.video.tune);
    }

    // 二次编码
    if (preset.video.passMode === 1 || preset.video.passMode === 2) {
      args.push('-pass', String(preset.video.passMode));
      if (preset.video.passLogFile) {
        args.push('-passlogfile', escapeShellArg(preset.video.passLogFile));
      }
    }

    // 质量控制
    const qualityArgs = this.buildQualityArgs(preset.quality);
    args.push(...qualityArgs);

    // 视频滤镜
    const filters = this.buildVideoFilters(preset);
    if (filters.length > 0) {
      args.push('-vf', filters.join(','));
    }

    // 像素格式 (从色彩管理中提取)
    if (preset.filters.colorManagement?.enabled && preset.filters.colorManagement.pixelFormat) {
      args.push('-pix_fmt', preset.filters.colorManagement.pixelFormat);
    }

    // 自定义视频参数
    if (preset.custom.videoParams) {
      args.push(...this.parseCustomParams(preset.custom.videoParams));
    }

    return args;
  }

  /**
   * 构建质量控制参数
   */
  private buildQualityArgs(quality: FFmpegPreset['quality']): string[] {
    const args: string[] = [];

    switch (quality.controlMode) {
      case 'CRF':
        args.push(`-${quality.paramName}`, quality.value);
        break;

      case 'VBR':
      case 'VBR_HQ':
        if (quality.bitrate.base) {
          args.push('-b:v', quality.bitrate.base);
        }
        if (quality.bitrate.min) {
          args.push('-minrate', quality.bitrate.min);
        }
        if (quality.bitrate.max) {
          args.push('-maxrate', quality.bitrate.max);
        }
        if (quality.bitrate.bufferSize) {
          args.push('-bufsize', quality.bitrate.bufferSize);
        }
        break;

      case 'CQP':
        args.push(`-${quality.paramName}`, quality.value);
        break;

      case 'CBR':
        if (quality.bitrate.base) {
          args.push('-b:v', quality.bitrate.base);
          args.push('-maxrate', quality.bitrate.base);
          args.push('-minrate', quality.bitrate.base);
          args.push('-bufsize', quality.bitrate.bufferSize || '2M');
        }
        break;
    }

    return args;
  }

  /**
   * 构建视频滤镜
   */
  private buildVideoFilters(preset: FFmpegPreset): string[] {
    const filters: string[] = [];
    const f = preset.filters;

    // 裁剪
    if (f.crop?.enabled) {
      filters.push(`crop=${f.crop.width}:${f.crop.height}:${f.crop.x}:${f.crop.y}`);
    }

    // 缩放
    if (f.scale?.enabled) {
      if (f.scale.keepAspect) {
        filters.push(`scale=${f.scale.width}:-1`);
      } else {
        filters.push(`scale=${f.scale.width}:${f.scale.height}`);
      }
    }

    // 帧率
    if (f.framerate?.enabled) {
      filters.push(`fps=${f.framerate.fps}`);
    }

    // 去隔行
    if (f.deinterlace?.enabled) {
      const deinterlaceFilter = this.getDeinterlaceFilter(f.deinterlace.mode);
      if (deinterlaceFilter) {
        filters.push(deinterlaceFilter);
      }
    }

    // 降噪
    if (f.denoise?.enabled) {
      const denoiseFilter = this.getDenoiseFilter(f.denoise.mode, f.denoise.strength);
      if (denoiseFilter) {
        filters.push(denoiseFilter);
      }
    }

    // 锐化
    if (f.sharpen?.enabled) {
      filters.push(`unsharp=${f.sharpen.strength}`);
    }

    // 字幕烧录
    if (f.subtitle?.enabled) {
      const subtitleFilter = this.getSubtitleFilter(f.subtitle);
      if (subtitleFilter) {
        filters.push(subtitleFilter);
      }
    }

    // 色彩管理 (不包括像素格式，单独处理)
    if (f.colorManagement?.enabled) {
      const colorFilters = this.getColorManagementFilters(f.colorManagement);
      filters.push(...colorFilters);
    }

    // 旋转/翻转
    if (f.transform?.enabled) {
      const transformFilters = this.getTransformFilters(f.transform);
      filters.push(...transformFilters);
    }

    // 自定义滤镜
    if (preset.custom.videoFilter) {
      filters.push(preset.custom.videoFilter);
    }

    return filters;
  }

  /**
   * 获取去隔行滤镜
   */
  private getDeinterlaceFilter(mode: number): string | null {
    const modes: Record<number, string> = {
      1: 'yadif=0:-1:0',  // 自动
      2: 'yadif=0:0:0',   // 顶部
      3: 'yadif=0:1:0',   // 底部
      4: 'tinterlace=4',   // TFF (Top Field First)
      5: 'tinterlace=6',   // BFF (Bottom Field First)
      6: 'fieldmatch,yadif=deint=interlaced,decimate', // IVTC
      7: 'yadif=1',        // 帧加倍
    };

    return modes[mode] || null;
  }

  /**
   * 获取降噪滤镜
   */
  private getDenoiseFilter(mode: string, strength: string): string | null {
    switch (mode) {
      case 'hqdn3d':
        return `hqdn3d=${strength}`;
      case 'nlmeans':
        return `nlmeans=s=${strength}`;
      case 'bm3d':
        return `bm3d=s=${strength}`;
      default:
        return null;
    }
  }

  /**
   * 获取字幕滤镜
   */
  private getSubtitleFilter(subtitle: SubtitleFilter): string | null {
    if (subtitle.source === 'external' && subtitle.file) {
      return `subtitles='${escapeShellArg(subtitle.file)}':force_style='${subtitle.styling?.forceStyle || ''}'`;
    } else if (subtitle.source === 'embedded' && subtitle.streamIndex !== undefined) {
      return `subtitles='${subtitle.streamIndex}':force_style='${subtitle.styling?.forceStyle || ''}'`;
    }
    return null;
  }

  /**
   * 获取色彩管理滤镜
   */
  private getColorManagementFilters(cm: ColorManagementFilter): string[] {
    const filters: string[] = [];

    if (cm.colorSpace) {
      filters.push(`colormatrix=${cm.colorSpace}`);
    }

    if (cm.transfer) {
      filters.push(`colorspace=${cm.transfer}`);
    }

    if (cm.primaries) {
      filters.push(`colorspace=primaries=${cm.primaries}`);
    }

    if (cm.range) {
      filters.push(`colorspace=range=${cm.range}`);
    }

    return filters;
  }

  /**
   * 获取变换滤镜
   */
  private getTransformFilters(transform: TransformFilter): string[] {
    const filters: string[] = [];

    if (transform.rotation && transform.rotation !== '0') {
      // 将角度转换为弧度
      const radians = (parseFloat(transform.rotation) * Math.PI) / 180;
      filters.push(`rotate=${radians}*PI/180`);
    }

    if (transform.flipH) {
      filters.push('hflip');
    }

    if (transform.flipV) {
      filters.push('vflip');
    }

    return filters;
  }

  /**
   * 构建音频编码参数
   */
  private buildAudioArgs(preset: FFmpegPreset): string[] {
    const args: string[] = [];

    // 编码器
    if (preset.audio.encoder) {
      args.push('-c:a', preset.audio.encoder);
    }

    // 比特率
    if (preset.audio.bitrate) {
      args.push('-b:a', preset.audio.bitrate);
    }

    // 声道数
    if (preset.audio.channels) {
      args.push('-ac', preset.audio.channels);
    }

    // 采样率
    if (preset.audio.sampleRate) {
      args.push('-ar', preset.audio.sampleRate);
    }

    // 强制编解码器
    if (preset.audio.codec) {
      args.push('-codec:a', preset.audio.codec);
    }

    // 自定义音频参数
    if (preset.custom.audioParams) {
      args.push(...this.parseCustomParams(preset.custom.audioParams));
    }

    return args;
  }

  /**
   * 构建流映射参数
   */
  private buildStreamMap(preset: FFmpegPreset): string[] {
    const args: string[] = [];

    if (preset.streamControl.mapAll) {
      // 映射所有流
      args.push('-map', '0');
    } else {
      // 选择性映射
      if (preset.video.enabled) {
        args.push('-map', '0:v:0');  // 主视频流
      }
      if (preset.audio.enabled) {
        args.push('-map', '0:a:0');  // 主音频流
      }
      if (preset.streamControl.keepSubtitleStreams) {
        args.push('-map', '0:s?');  // 所有字幕流
      }
    }

    return args;
  }

  /**
   * 添加解码参数
   */
  private addDecoderArgs(args: string[], preset: FFmpegPreset): void {
    if (preset.decoder.decoder && preset.decoder.decoder !== 'auto') {
      args.push('-c:v', preset.decoder.decoder);
    }
    if (preset.decoder.hwaccel && preset.decoder.hwaccel !== 'auto') {
      args.push('-hwaccel', preset.decoder.hwaccel);
    }
    if (preset.decoder.hwaccelDevice) {
      args.push('-hwaccel_device', preset.decoder.hwaccelDevice);
    }
  }

  /**
   * 添加剪辑参数
   */
  private addTrimmingArgs(args: string[], preset: FFmpegPreset): void {
    if (preset.trimming.enabled) {
      if (preset.trimming.startTime) {
        args.push('-ss', TimeUtils.timeToSeconds(preset.trimming.startTime));
      }
      if (preset.trimming.endTime) {
        args.push('-to', TimeUtils.timeToSeconds(preset.trimming.endTime));
      }
    }
  }

  /**
   * 添加元数据参数
   */
  private addMetadataArgs(args: string[], metadata: MetadataConfig): void {
    if (metadata.title) {
      args.push('-metadata', `title=${metadata.title}`);
    }
    if (metadata.artist) {
      args.push('-metadata', `artist=${metadata.artist}`);
    }
    if (metadata.album) {
      args.push('-metadata', `album=${metadata.album}`);
    }
    if (metadata.year) {
      args.push('-metadata', `year=${metadata.year}`);
    }
    if (metadata.comment) {
      args.push('-metadata', `comment=${metadata.comment}`);
    }
  }

  /**
   * 解析自定义参数
   */
  private parseCustomParams(params: string): string[] {
    return params
      .split(/\s+/)
      .filter(p => p.trim().length > 0);
  }

  /**
   * 生成批量命令
   */
  async buildBatch(
    preset: FFmpegPreset,
    inputFiles: string[],
    outputGenerator?: (inputFile: string) => string
  ): Promise<string[]> {
    const commands: string[] = [];

    for (const inputFile of inputFiles) {
      const outputFile = outputGenerator
        ? outputGenerator(inputFile)
        : this.generateOutputPath(inputFile, preset);

      const command = await this.build(preset, inputFile, outputFile);
      commands.push(command);
    }

    return commands;
  }

  /**
   * 生成输出路径
   */
  private generateOutputPath(inputFile: string, preset: FFmpegPreset): string {
    const path = require('path');
    const parsed = path.parse(inputFile);

    if (preset.output.naming.useAutoNaming) {
      let name = parsed.name;

      // 添加前缀
      if (preset.output.naming.prefix) {
        name = preset.output.naming.prefix + name;
      }

      // 添加后缀
      if (preset.output.naming.suffix) {
        name += preset.output.naming.suffix;
      }

      // 自定义模式
      if (preset.output.naming.customPattern) {
        name = this.applyCustomNamingPattern(
          preset.output.naming.customPattern,
          parsed.name,
          parsed.dir
        );
      }

      return path.join(parsed.dir, `${name}.${preset.output.container}`);
    } else {
      return preset.output.location || path.join(parsed.dir, `${parsed.name}.${preset.output.container}`);
    }
  }

  /**
   * 应用自定义命名模式
   */
  private applyCustomNamingPattern(pattern: string, filename: string, dir: string): string {
    const now = new Date();
    return pattern
      .replace('{name}', filename)
      .replace('{date}', now.toISOString().split('T')[0])
      .replace('{time}', now.toTimeString().split(' ')[0])
      .replace('{timestamp}', String(now.getTime()));
  }
}

export const commandBuilder = new CommandBuilder();
```

#### 3.4.2 ValidationService 类

```typescript
// src/ffmpeg/services/validationService.ts

import type { FFmpegPreset } from '../types/preset';

/**
 * 验证结果
 */
export interface ValidationResult {
  valid: boolean;
  errors: string[];
  warnings: string[];
}

/**
 * FFmpeg 参数验证服务
 */
export class ValidationService {
  /**
   * 验证完整预设
   */
  validatePreset(preset: FFmpegPreset): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证输出配置
    const outputValidation = this.validateOutputConfig(preset.output);
    errors.push(...outputValidation.errors);
    warnings.push(...outputValidation.warnings);

    // 验证视频配置
    if (preset.video.enabled) {
      const videoValidation = this.validateVideoConfig(preset.video);
      errors.push(...videoValidation.errors);
      warnings.push(...videoValidation.warnings);
    }

    // 验证音频配置
    if (preset.audio.enabled) {
      const audioValidation = this.validateAudioConfig(preset.audio);
      errors.push(...audioValidation.errors);
      warnings.push(...audioValidation.warnings);
    }

    // 验证质量配置
    const qualityValidation = this.validateQualityConfig(preset.quality);
    errors.push(...qualityValidation.errors);
    warnings.push(...qualityValidation.warnings);

    // 验证滤镜配置
    const filterValidation = this.validateFilterConfig(preset.filters);
    errors.push(...filterValidation.errors);
    warnings.push(...filterValidation.warnings);

    // 验证剪辑配置
    if (preset.trimming.enabled) {
      const trimmingValidation = this.validateTrimmingConfig(preset.trimming);
      errors.push(...trimmingValidation.errors);
      warnings.push(...trimmingValidation.warnings);
    }

    return {
      valid: errors.length === 0,
      errors,
      warnings,
    };
  }

  /**
   * 验证输出配置
   */
  private validateOutputConfig(output: FFmpegPreset['output']): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证容器格式
    if (!output.container) {
      errors.push('Output container is required');
    }

    // 验证命名模式
    if (output.naming.useAutoNaming && output.naming.customPattern) {
      if (!this.validateNamingPattern(output.naming.customPattern)) {
        errors.push('Invalid custom naming pattern');
      }
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证视频配置
   */
  private validateVideoConfig(video: FFmpegPreset['video']): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证编码器
    if (!video.encoder) {
      errors.push('Video encoder is required when video is enabled');
    }

    // 验证二次编码配置
    if (video.passMode === 2 && !video.passLogFile) {
      warnings.push('Pass log file is recommended for two-pass encoding');
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证音频配置
   */
  private validateAudioConfig(audio: FFmpegPreset['audio']): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证编码器
    if (!audio.encoder) {
      errors.push('Audio encoder is required when audio is enabled');
    }

    // 验证比特率格式
    if (audio.bitrate && !this.validateBitrate(audio.bitrate)) {
      errors.push('Invalid audio bitrate format');
    }

    // 验证采样率
    if (audio.sampleRate && !this.validateSampleRate(audio.sampleRate)) {
      errors.push('Invalid audio sample rate');
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证质量配置
   */
  private validateQualityConfig(quality: FFmpegPreset['quality']): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证 CRF 范围
    if (quality.controlMode === 'CRF') {
      const crf = parseFloat(quality.value);
      if (isNaN(crf) || crf < 0 || crf > 51) {
        errors.push('CRF value must be between 0 and 51');
      }
    }

    // 验证比特率格式
    if (['VBR', 'VBR_HQ', 'CBR'].includes(quality.controlMode)) {
      if (!quality.bitrate.base) {
        errors.push('Bitrate is required for VBR/CBR mode');
      }
      if (!this.validateBitrate(quality.bitrate.base)) {
        errors.push('Invalid bitrate format');
      }
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证滤镜配置
   */
  private validateFilterConfig(filters: FFmpegPreset['filters']): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证裁剪
    if (filters.crop?.enabled) {
      if (!this.validateCrop(filters.crop)) {
        errors.push('Invalid crop parameters');
      }
    }

    // 验证缩放
    if (filters.scale?.enabled) {
      if (!this.validateScale(filters.scale)) {
        errors.push('Invalid scale parameters');
      }
    }

    // 验证字幕
    if (filters.subtitle?.enabled) {
      if (filters.subtitle.source === 'external' && !filters.subtitle.file) {
        errors.push('External subtitle file path is required');
      }
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证剪辑配置
   */
  private validateTrimmingConfig(trimming: FFmpegPreset['trimming']): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 验证时间格式
    if (!this.validateTimeFormat(trimming.startTime)) {
      errors.push('Invalid start time format');
    }

    if (trimming.endTime && !this.validateTimeFormat(trimming.endTime)) {
      errors.push('Invalid end time format');
    }

    // 验证时间顺序
    if (trimming.startTime && trimming.endTime) {
      const start = this.timeToSeconds(trimming.startTime);
      const end = this.timeToSeconds(trimming.endTime);
      if (start >= end) {
        errors.push('Start time must be before end time');
      }
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证比特率格式
   */
  private validateBitrate(bitrate: string): boolean {
    return /^\d+[kMG]?$/i.test(bitrate);
  }

  /**
   * 验证采样率
   */
  private validateSampleRate(sampleRate: string): boolean {
    const validRates = ['8000', '11025', '22050', '44100', '48000', '96000'];
    return validRates.includes(sampleRate);
  }

  /**
   * 验证时间格式
   */
  private validateTimeFormat(time: string): boolean {
    return /^\d{2}:\d{2}:\d{2}(\.\d{2,3})?$/.test(time);
  }

  /**
   * 验证命名模式
   */
  private validateNamingPattern(pattern: string): boolean {
    // 基本验证：不包含非法字符
    return /[<>:"/\\|?*]/.test(pattern) === false;
  }

  /**
   * 验证裁剪参数
   */
  private validateCrop(crop: any): boolean {
    return (
      crop.width &&
      crop.height &&
      crop.x !== undefined &&
      crop.y !== undefined
    );
  }

  /**
   * 验证缩放参数
   */
  private validateScale(scale: any): boolean {
    return scale.width && scale.height;
  }

  /**
   * 时间转秒数
   */
  private timeToSeconds(time: string): number {
    const [hours, minutes, seconds] = time.split(':').map(parseFloat);
    return hours * 3600 + minutes * 60 + seconds;
  }

  /**
   * 验证视频参数数组
   */
  validateVideoArgs(args: string[]): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 检查必需参数
    if (!args.includes('-c:v') && !args.includes('-codec:v')) {
      warnings.push('Video codec not specified');
    }

    // 检查输出格式
    if (!args.includes('-pix_fmt')) {
      warnings.push('Pixel format not specified, may use default');
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证音频参数数组
   */
  validateAudioArgs(args: string[]): ValidationResult {
    const errors: string[] = [];
    const warnings: string[] = [];

    // 检查必需参数
    if (!args.includes('-c:a') && !args.includes('-codec:a')) {
      warnings.push('Audio codec not specified');
    }

    return { valid: errors.length === 0, errors, warnings };
  }

  /**
   * 验证滤镜链
   */
  validateFilterChain(filters: string[]): boolean {
    if (filters.length === 0) return true;

    // 检查滤镜格式
    const filterRegex = /^[a-zA-Z0-9_]+=?.*$/;
    return filters.every(f => filterRegex.test(f));
  }
}
```

### 4.4 核心类实现

```typescript
// src/ffmpeg/services/progressParser.ts

import type {
  FFmpegProgress,
  FFmpegProgressEvent,
  ProgressPatterns
} from '../types/progress';

/**
 * FFmpeg 进度解析器
 * 参考自 FFmpegFreeUI/编码任务/编码任务.vb
 */
export class ProgressParser {
  private duration: number | null = null;
  private progress: Partial<FFmpegProgress> = {};

  // 正则表达式模式
  private readonly patterns: ProgressPatterns = {
    duration: /Duration:\s*(\d{2}:\d{2}:\d{2}\.\d{2,3})/,
    frame: /frame=\s*(\d+)/,
    fps: /fps=\s*(\d+\.?\d*)/,
    q: /q=\s*([\d\.]+)/,
    size: /size=\s*(\d+)\s*([KMG]iB)/,
    time: /time=\s*(\d{2}:\d{2}:\d{2}\.\d{2,3})/,
    bitrate: /bitrate:\s*([\d\.]+)\s*kbits\/s/,
    speed: /speed=\s*([\d\.eE\+\-]+)\s*x/,
    error: /Error\s+(.+)/i,
    warning: /Warning\s+(.+)/i,
  };

  // 错误关键词列表
  private readonly errorKeywords = [
    'Conversion failed',
    'Invalid data',
    'Operation not permitted',
    'No such file or directory',
    'Permission denied',
  ];

  /**
   * 解析一行输出
   * @param line FFmpeg 输出的一行文本
   * @returns 进度事件或 null
   */
  parse(line: string): FFmpegProgressEvent | null {
    // 解析时长
    const durationMatch = line.match(this.patterns.duration);
    if (durationMatch) {
      this.duration = this.parseTimeString(durationMatch[1]);
      return {
        type: 'duration',
        duration: durationMatch[1],
        durationSeconds: this.duration,
      };
    }

    // 解析错误
    const errorMatch = line.match(this.patterns.error);
    if (errorMatch) {
      return {
        type: 'error',
        error: errorMatch[1],
      };
    }

    // 解析警告
    const warningMatch = line.match(this.patterns.warning);
    if (warningMatch) {
      return {
        type: 'warning',
        warning: warningMatch[1],
      };
    }

    // 检查错误关键词
    for (const keyword of this.errorKeywords) {
      if (line.includes(keyword)) {
        return {
          type: 'error',
          error: line.trim(),
        };
      }
    }

    // 解析进度
    const progressData = this.extractProgressData(line);
    if (progressData && Object.keys(progressData).length > 0) {
      this.progress = { ...this.progress, ...progressData };

      // 计算衍生信息
      this.calculateDerivedMetrics();

      return {
        type: 'progress',
        data: this.progress as FFmpegProgress,
      };
    }

    return null;
  }

  /**
   * 提取进度数据
   */
  private extractProgressData(line: string): Partial<FFmpegProgress> | null {
    const data: Partial<FFmpegProgress> = {};
    let hasData = false;

    // 帧
    const frameMatch = line.match(this.patterns.frame);
    if (frameMatch) {
      data.frame = parseInt(frameMatch[1]);
      hasData = true;
    }

    // 帧率
    const fpsMatch = line.match(this.patterns.fps);
    if (fpsMatch) {
      data.fps = parseFloat(fpsMatch[1]);
      hasData = true;
    }

    // 量化参数
    const qMatch = line.match(this.patterns.q);
    if (qMatch) {
      data.q = parseFloat(qMatch[1]);
      hasData = true;
    }

    // 大小
    const sizeMatch = line.match(this.patterns.size);
    if (sizeMatch) {
      data.size = parseInt(sizeMatch[1]);
      data.sizeUnit = sizeMatch[2] as 'KiB' | 'MiB' | 'GiB';
      data.sizeBytes = this.convertSizeToBytes(data.size, data.sizeUnit);
      hasData = true;
    }

    // 时间
    const timeMatch = line.match(this.patterns.time);
    if (timeMatch) {
      data.time = timeMatch[1];
      data.timeSeconds = this.parseTimeString(timeMatch[1]);
      hasData = true;
    }

    // 比特率
    const bitrateMatch = line.match(this.patterns.bitrate);
    if (bitrateMatch) {
      data.bitrate = parseFloat(bitrateMatch[1]);
      hasData = true;
    }

    // 速度
    const speedMatch = line.match(this.patterns.speed);
    if (speedMatch) {
      data.speed = parseFloat(speedMatch[1]);
      hasData = true;
    }

    return hasData ? data : null;
  }

  /**
   * 计算衍生指标
   */
  private calculateDerivedMetrics(): void {
    // 计算进度百分比
    if (this.progress.timeSeconds && this.duration) {
      this.progress.progress = Math.min(
        (this.progress.timeSeconds / this.duration) * 100,
        100
      );

      // 计算预估大小
      if (this.progress.sizeBytes && this.progress.progress) {
        this.progress.estimatedSize = Math.round(
          this.progress.sizeBytes / (this.progress.progress / 100)
        );
      }

      // 计算剩余时间
      if (this.progress.speed && this.progress.timeSeconds) {
        const remaining = (this.duration - this.progress.timeSeconds) / this.progress.speed;
        this.progress.remainingTime = Math.max(0, Math.round(remaining));
      }
    }
  }

  /**
   * 解析时间字符串
   * 格式: HH:MM:SS.mmm
   */
  parseTimeString(timeStr: string): number {
    const parts = timeStr.split(':');
    if (parts.length !== 3) return 0;

    const [hours, minutes, seconds] = parts.map(parseFloat);
    return hours * 3600 + minutes * 60 + seconds;
  }

  /**
   * 转换大小到字节
   */
  convertSizeToBytes(
    value: number,
    unit: 'KiB' | 'MiB' | 'GiB'
  ): number {
    switch (unit) {
      case 'KiB':
        return value * 1024;
      case 'MiB':
        return value * 1024 * 1024;
      case 'GiB':
        return value * 1024 * 1024 * 1024;
      default:
        return value;
    }
  }

  /**
   * 获取当前时长
   */
  getDuration(): number | null {
    return this.duration;
  }

  /**
   * 获取当前进度
   */
  getProgress(): Partial<FFmpegProgress> {
    return this.progress;
  }

  /**
   * 重置解析器状态
   */
  reset(): void {
    this.duration = null;
    this.progress = {};
  }

  /**
   * 解析完成事件
   */
  parseComplete(exitCode: number): FFmpegProgressEvent {
    return {
      type: 'complete',
      exitCode,
    };
  }
}

// 导出单例
export const progressParser = new ProgressParser();
```

```typescript
function handleOutput(ptyId: string, data: string) {
  const tab = tabs.value.find(t => t.ptyId === ptyId);

  // 常规输出处理
  tab.output += data;

  // FFmpeg 进度解析

    const event = progressParser.parse(data);
    if (event) {
      handleFFmpegProgressEvent(tab, event);
    }
  }
}

function handleFFmpegProgressEvent(tab: TerminalTab, event: FFmpegProgressEvent) {
  switch (event.type) {
    case 'progress':
      tab.ffmpegProgress = event.data;
      break;
    case 'duration':
      // 存储总时长
      break;
    case 'complete':
      if (event.exitCode === 0) {
        tab.status = 'idle';
      } else {
        tab.status = 'error';
      }
      break;
    case 'error':
      tab.status = 'error';
      tab.error = event.error;
      break;
    case 'warning':
      // 记录警告
      console.warn('[FFmpeg]', event.warning);
      break;
  }
```

```typescript
import { defineStore } from 'pinia';
import { ref } from 'vue';
import type {
  FFmpegPreset,
  PresetMetadata,
  PresetEntry,
  FFmpegFreeUIPreset
} from '../types/preset';
import { PresetConverter } from '../services/presetConverter';
import { safeInvoke } from '../../utils/programUtils';

export const useFFmpegPresetsStore = defineStore('ffmpegPresets', () => {
  // ==================== State ====================

  const presets = ref<Map<string, PresetEntry>>(new Map());
  const builtinPresets = ref<FFmpegPreset[]>([]);
  const selectedPresetId = ref<string | null>(null);

  const presetConverter = new PresetConverter();

  // ==================== Actions ====================

  /**
   * 加载内置预设
   */
  async function loadBuiltinPresets() {
    const { builtinPresets: presets } = await import('../data/builtinPresets');
    builtinPresets.value = presets;
  }

  /**
   * 保存预设
   */
  async function savePreset(
    preset: FFmpegPreset,
    name: string,
    description?: string,
    tags?: string[]
  ) {
    const meta: PresetMetadata = {
      id: `preset-${Date.now()}`,
      name,
      description,
      version: '1.0',
      createdAt: Date.now(),
      updatedAt: Date.now(),
      author: '',
      tags,
    };

    const key = `ffmpeg.preset.${meta.id}`;
    const value = JSON.stringify({ preset, meta });

    // 通过 Node.js 存储保存
    await safeInvoke('save_preset', { key, value });

    presets.value.set(meta.id, { preset, meta });
    selectedPresetId.value = meta.id;
  }

  /**
   * 加载预设
   */
  async function loadPreset(id: string): Promise<FFmpegPreset | null> {
    const key = `ffmpeg.preset.${id}`;
    const value = await safeInvoke<string>('load_preset', { key });

    if (value) {
      const { preset, meta } = JSON.parse(value);
      presets.value.set(id, { preset, meta });
      return preset;
    }

    return null;
  }

  /**
   * 删除预设
   */
  async function deletePreset(id: string) {
    const key = `ffmpeg.preset.${id}`;
    await safeInvoke('delete_preset', { key });
    presets.value.delete(id);

    if (selectedPresetId.value === id) {
      selectedPresetId.value = null;
    }
  }

  /**
   * 导出预设到文件
   */
  async function exportPresetToFile(
    preset: FFmpegPreset,
    filePath: string,
    format: 'json' | '3fui' = 'json'
  ) {
    let content: string;

    if (format === '3fui') {
      const oldPreset = presetConverter.convertRebebucaTo3FUI(preset);
      content = JSON.stringify(oldPreset, null, 2);
    } else {
      content = JSON.stringify(preset, null, 2);
    }

    await safeInvoke('write_text_file', { path: filePath, content });
  }

   * 从文件导入预设
   */
  async function importPresetFromFile(filePath: string): Promise<FFmpegPreset> {
    const content = await safeInvoke<string>('read_text_file', { path: filePath });
    return JSON.parse(content);
  }

  /**
   * 从 3FUI 格式导入
   */
  async function importFrom3FUI(filePath: string): Promise<FFmpegPreset> {
    const content = await safeInvoke<string>('read_text_file', { path: filePath });
    const oldPreset: FFmpegFreeUIPreset = JSON.parse(content);

    return presetConverter.convert3FUIToRebebuca(oldPreset);
  }

  /**
   * 导出为 3FUI 格式
   */
  async function exportTo3FUI(preset: FFmpegPreset, filePath: string) {
    return exportPresetToFile(preset, filePath, '3fui');
  }

  /**
   * 批量导入预设
   */
  async function importPresetsFromDirectory(dirPath: string): Promise<FFmpegPreset[]> {
    const files = await safeInvoke<string[]>('list_files', {
      path: dirPath,
      pattern: '*.{json,3fui}',

    const importedPresets: FFmpegPreset[] = [];

    for (const file of files) {
      try {
        if (file.endsWith('.3fui')) {
          const preset = await importFrom3FUI(file);
          importedPresets.push(preset);
        } else if (file.endsWith('.json')) {
          const preset = await importPresetFromFile(file);
          importedPresets.push(preset);
        }
      } catch (error) {
        console.error(`Failed to import preset from ${file}:`, error);
      }
    }

    return importedPresets;
  }

  /**
   * 获取预设列表
   */
  function getPresetList(): PresetMetadata[] {
    return Array.from(presets.value.values()).map(entry => entry.meta);
  }

  /**
   * 搜索预设
   */
  function searchPresets(query: string): PresetMetadata[] {
    const lowerQuery = query.toLowerCase();
    return getPresetList().filter(meta =>
      meta.name.toLowerCase().includes(lowerQuery) ||
      meta.description?.toLowerCase().includes(lowerQuery) ||
      meta.tags?.some(tag => tag.toLowerCase().includes(lowerQuery))
    );
  }

  /**
   * 初始化加载所有预设
   */
  async function initialize() {
    await loadBuiltinPresets();

    // 加载用户预设
    try {
      const presetKeys = await safeInvoke<string[]>('list_preset_keys', {
        prefix: 'ffmpeg.preset.',
      });

      for (const key of presetKeys) {
        const id = key.replace('ffmpeg.preset.', '');
        await loadPreset(id);
      }
    } catch (error) {
      console.error('Failed to load user presets:', error);
    }
  }

  return {
    // State
    presets,
    builtinPresets,
    selectedPresetId,

    // Actions
    savePreset,
    loadPreset,
    deletePreset,
    exportPresetToFile,
    importPresetFromFile,
    importFrom3FUI,
    exportTo3FUI,
    importPresetsFromDirectory,
    getPresetList,
    searchPresets,
    initialize,
  };
});
```

```typescript
    // 输出配置
    old.输出容器 = preset.output.container;
    old.输出命名_使用自动命名 = preset.output.naming.useAutoNaming;

    old.视频参数_编码器_具体编码 = preset.video.encoder;

    old.视频参数_编码器_配置文件 = preset.video.profile;
    old.视频参数_编码器_级别 = preset.video.level;

      old.视频参数_裁剪_X坐标 = preset.filters.crop.x;

    const paramName = this.parseQualityParamName(old.视频参数_质量控制_参数名);

  private parseQualityControlMode(mode?: string): QualityControlMode {
    const modeMap: Record<string, QualityControlMode> = {
      'CRF': 'CRF',
      'VBR': 'VBR',
      'VBR HQ': 'VBR_HQ',
      'CQP': 'CQP',
      'CBR': 'CBR',

    return modeMap[mode || ''] || 'CRF';

   */

    if (!param) return 'crf';
    const paramMap: Record<string, QualityParamName> = {
      'crf': 'crf',
      'cq': 'cq',
      'qp': 'qp',
      'global_quality': 'global_quality',
    };

    return paramMap[param] || 'crf';
  }

    if (old.视频参数_裁剪_启用) {
      filters.crop = {
        enabled: true,

      filters.scale = {
        enabled: true,

    }

    // 去隔行
    if (old.视频参数_去隔行) {
      filters.deinterlace = {
        enabled: true,
        mode: old.视频参数_去隔行_模式 || 1,
      };
    }

    // 字幕
    if (old.视频参数_烧录字幕_启用) {
      filters.subtitle = {
        enabled: true,
        source: old.视频参数_烧录字幕_来源 || 'embedded',

          outlineColor: old.视频参数_烧录字幕_边框颜色,
          alignment: old.视频参数_烧录字幕_对齐方式,
        },
      };
    }

    // 色彩管理
    if (old.视频参数_色彩管理_启用) {
      filters.colorManagement = {
        enabled: true,
        pixelFormat: old.视频参数_色彩管理_像素格式,

        transfer: old.视频参数_色彩管理_传输特性,
        primaries: old.视频参数_色彩管理_原色,
        range: old.视频参数_色彩管理_范围,
      };
    }

    return filters;
  }
```

```typescript
// src/ffmpeg/utils/time.ts

/**
 * 时间转换工具
 */
export class TimeUtils {
  /**
   * 时间字符串转秒数
   * @param timeStr 时间字符串 (HH:MM:SS.mmm)
   * @returns 秒数
   */
  static timeToSeconds(timeStr: string): number {
    const parts = timeStr.split(':');
    if (parts.length !== 3) return 0;

    const [hours, minutes, seconds] = parts.map(parseFloat);
    return hours * 3600 + minutes * 60 + seconds;
  }

  /**
   * 秒数转时间字符串
   * @param seconds 秒数
   * @param decimalPlaces 小数位数 (默认 2)
   * @returns 时间字符串 (HH:MM:SS.mmm)
   */
  static secondsToTime(seconds: number, decimalPlaces: number = 2): string {
    const hours = Math.floor(seconds / 3600);
    const minutes = Math.floor((seconds % 3600) / 60);
    const secs = seconds % 60;

    const pad = (num: number, size: number): string => {
      return num.toString().padStart(size, '0');
    };

    const secsStr = secs.toFixed(decimalPlaces);
    const [intPart, decPart = ''] = secsStr.split('.');

    return `${pad(hours, 2)}:${pad(minutes, 2)}:${pad(intPart, 2)}${decPart ? `.${decPart}` : ''}`;
  }

  /**
   * 格式化时长
   * @param seconds 秒数
   * @returns 格式化的时长字符串 (例如: "1h 23m 45s")
   */
  static formatDuration(seconds: number): string {
    const hours = Math.floor(seconds / 3600);
    const minutes = Math.floor((seconds % 3600) / 60);
    const secs = Math.floor(seconds % 60);

    const parts: string[] = [];

    if (hours > 0) {
      parts.push(`${hours}h`);
    }
    if (minutes > 0) {
      parts.push(`${minutes}m`);
    }
    if (secs > 0 || parts.length === 0) {
      parts.push(`${secs}s`);
    }

    return parts.join(' ');
  }

  /**
   * 解析 FFmpeg 时间字符串
   * @param timeStr FFmpeg 时间字符串 (例如: "01:23:45.678")
   * @returns 毫秒数
   */
  static parseFFmpegTime(timeStr: string): number {
    const seconds = this.timeToSeconds(timeStr);
    return Math.round(seconds * 1000);
  }
}

export default TimeUtils;
```

```typescript
// src/ffmpeg/utils/path.ts

import path from 'path';

/**
 * 路径工具
 */
export class PathUtils {
  /**
   * 标准化路径
   * @param filePath 文件路径
   * @returns 标准化后的路径
   */
  static normalizePath(filePath: string): string {
    return path.normalize(filePath).replace(/\\/g, '/');
  }

  /**
   * 转义 Shell 参数
   * @param arg 参数值
   * @returns 转义后的参数
   */
  static escapeShellArg(arg: string): string {
    // 使用单引号包裹，单引号用 '\'' 替换
    const escaped = arg.replace(/'/g, "'\\''");
    return `'${escaped}'`;
  }

  /**
   * 转义 Windows 路径
   * @param filePath Windows 文件路径
   * @returns 转义后的路径
   */
  static escapeWindowsPath(filePath: string): string {
    // 将反斜杠替换为正斜杠
    return filePath.replace(/\\/g, '/');
  }

  /**
   * 生成输出文件路径
   * @param inputFile 输入文件路径
   * @param container 输出容器格式
   * @param naming 命名配置
   * @returns 输出文件路径
   */
  static generateOutputPath(
    inputFile: string,
    container: string,
    naming: {
      useAutoNaming: boolean;
      prefix?: string;
      suffix?: string;
      customPattern?: string;
    }
  ): string {
    const parsed = path.parse(this.normalizePath(inputFile));
    let name = parsed.name;

    if (naming.useAutoNaming) {
      // 添加前缀
      if (naming.prefix) {
        name = naming.prefix + name;
      }

      // 添加后缀
      if (naming.suffix) {
        name += naming.suffix;
      }

      // 自定义模式
      if (naming.customPattern) {
        name = this.applyCustomNamingPattern(naming.customPattern, parsed.name);
      }
    }

    return path.join(parsed.dir, `${name}.${container}`);
  }

  /**
   * 应用自定义命名模式
   * @param pattern 命名模式 (支持占位符: {name}, {date}, {time}, {timestamp})
   * @param filename 原始文件名
   * @returns 生成的文件名
   */
  static applyCustomNamingPattern(pattern: string, filename: string): string {
    const now = new Date();
    return pattern
      .replace('{name}', filename)
      .replace('{date}', now.toISOString().split('T')[0])
      .replace('{time}', now.toTimeString().split(' ')[0])
      .replace('{timestamp}', String(now.getTime()));
  }

  /**
   * 验证文件路径
   * @param filePath 文件路径
   * @returns 是否有效
   */
  static isValidPath(filePath: string): boolean {
    // 检查非法字符
    const invalidChars = /[<>:"|?*]/;
    return !invalidChars.test(filePath);
  }

  /**
   * 获取文件扩展名
   * @param filePath 文件路径
   * @returns 文件扩展名 (不含点)
   */
  static getExtension(filePath: string): string {
    const ext = path.extname(filePath);
    return ext ? ext.substring(1).toLowerCase() : '';
  }
}

export const escapeShellArg = PathUtils.escapeShellArg.bind(PathUtils);
export const normalizePath = PathUtils.normalizePath.bind(PathUtils);
export default PathUtils;
```

```typescript
import type { FFmpegTask } from './ffmpegTask';

/**
 * 执行 FFmpeg 任务
 */
export async function executeFFmpegTask(task: FFmpegTask): Promise<void> {
  const terminalStore = useTerminalStore();

  // 生成命令行
  const command = await commandBuilder.build(
    task.ffmpegConfig,
    task.inputFiles[0],
    task.outputFile || generateOutputPath(task.inputFiles[0], task.ffmpegConfig)
  );

  // 解析命令
  const parsed = parseCommandLine(command);

  // 创建终端 tab
  const tab = await terminalStore.createTab({
    title: task.name,
    command: parsed.command,
    args: parsed.args,
    cwd: task.cwd,
    isFFmpegTask: true,
  });

  // 标记为 FFmpeg 任务
  task.terminalTabId = tab.id;
}
```
