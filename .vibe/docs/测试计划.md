**示例测试代码**:

```typescript
import { describe, it, expect } from 'vitest';
import { commandBuilder } from '../commandBuilder';
import { createDefaultPreset } from '../../utils/testUtils';

describe('CommandBuilder', () => {
  describe('buildVideoArgs', () => {
    it('CB-001: should generate H.264 encoding command', async () => {
      const preset = createDefaultPreset();
      preset.video.encoder = 'libx264';
      preset.video.preset = 'medium';

      const command = await commandBuilder.build(preset, 'input.mp4', 'output.mp4');

      expect(command).toContain('-c:v libx264');
      expect(command).toContain('-preset medium');
    });

    it('CB-005: should generate CRF quality control', async () => {
      const preset = createDefaultPreset();
      preset.quality.controlMode = 'CRF';
      preset.quality.value = '23';
      preset.quality.paramName = 'crf';

      const command = await commandBuilder.build(preset, 'input.mp4', 'output.mp4');

      expect(command).toContain('-crf 23');
    });

    it('CB-011: should generate crop filter', async () => {
      const preset = createDefaultPreset();
      preset.filters.crop = {
        enabled: true,
        width: '1920',
        height: '1080',
        x: '0',
        y: '0',
      };

      const command = await commandBuilder.build(preset, 'input.mp4', 'output.mp4');

      expect(command).toContain('-vf crop=1920:1080:0:0');
    });

    it('CB-026: should escape file paths with spaces', async () => {
      const preset = createDefaultPreset();
      const inputFile = 'input with spaces.mp4';
      const outputFile = 'output with spaces.mp4';

      const command = await commandBuilder.build(preset, inputFile, outputFile);

      expect(command).toContain("'input with spaces.mp4'");
      expect(command).toContain("'output with spaces.mp4'");
    });
  });

  describe('buildBatch', () => {
    it('CB-027: should generate multiple commands for batch processing', async () => {
      const preset = createDefaultPreset();
      const inputFiles = ['file1.mp4', 'file2.mp4', 'file3.mp4'];

      const commands = await commandBuilder.buildBatch(preset, inputFiles);

      expect(commands).toHaveLength(3);
      expect(commands[0]).toContain('file1.mp4');
      expect(commands[1]).toContain('file2.mp4');
      expect(commands[2]).toContain('file3.mp4');
    });
  });
});
```

# .github/workflows/ffmpeg-test.yml
name: FFmpeg Module Tests

on:
  pull_request:
    paths:
      - 'src/ffmpeg/**'
      - 'src/stores/*ffmpeg*'
  push:
    branches: [main]

jobs:
  unit-tests:
    runs-on: ${{ matrix.os }}
    strategy:
      matrix:
        os: [ubuntu-latest, windows-latest, macos-latest]
        node-version: [18.x]

    steps:
      - uses: actions/checkout@v4
      - name: Setup pnpm
        uses: pnpm/action-setup@v2
        with:
          version: 9
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: ${{ matrix.node-version }}
          cache: 'pnpm'
      - name: Install dependencies
        run: pnpm install
      - name: Install FFmpeg
        run: |
          if [ "$RUNNER_OS" == "Linux" ]; then
            sudo apt-get update && sudo apt-get install -y ffmpeg
          elif [ "$RUNNER_OS" == "macOS" ]; then
            brew install ffmpeg
          fi
        shell: bash
      - name: Run unit tests
        run: pnpm test:unit:ffmpeg
      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          files: ./coverage/ffmpeg/lcov.info
          flags: ffmpeg-unit

  e2e-tests:
    runs-on: ${{ matrix.os }}
    strategy:
      matrix:
        os: [ubuntu-latest, windows-latest]
        node-version: [18.x]

    steps:
      - uses: actions/checkout@v4
      - name: Setup pnpm

        run: pnpm install
      - name: Install FFmpeg
        run: |
          if [ "$RUNNER_OS" == "Linux" ]; then
            sudo apt-get update && sudo apt-get install -y ffmpeg
          elif [ "$RUNNER_OS" == "Windows" ]; then
            choco install ffmpeg
          fi
        shell: bash
      - name: Install Playwright browsers
        run: npx playwright install --with-deps
      - name: Build Node.js app
        run: pnpm node.js build
      - name: Run E2E tests
        run: pnpm test:e2e:ffmpeg
      - name: Upload E2E screenshots

import { execSync } from 'child_process';
import { writeFileSync } from 'fs';

function generateTestReport() {

    <html>

      <title>FFmpeg Module Test Report</title>

        <p>Failed: ${unitTestResult.numFailedTests}</p>

      </div>

}

generateTestReport();

## 八、风险管理

### 8.1 测试风险

| 风险 | 可能性 | 影响 | 应对措施 |
|------|-------|------|---------|

| **硬件资源不足** | 中 | 中 | 使用云测试平台、分批测试 |
| **FFmpeg 版本差异** | 高 | 中 | 测试多个 FFmpeg 版本、兼容性处理 |
| **自动化测试不稳定** | 中 | 中 | 增加 flakiness 处理、重试机制 |
| **回归测试遗漏** | 中 | 高 | 建立完整的回归测试套件 |
| **用户反馈延迟** | 中 | 中 | 提前邀请 Beta 测试用户 |
